## 6. Independent verification and repeat FULL

Paste the entire block into PowerShell. Default is **read-only** (28 checks). To test repeat FULL and overwrite on the dedicated test PDBs, set `RUN_REPEAT = True` before executing. This commits one TARGET location marker, runs FULL, compares data again and tests an identity insert with rollback. If FULL fails, inspect TARGET; the marker or partial changes may remain. The identity generator can advance even after rollback. SOURCE is not modified.

```powershell
@'
import sys, subprocess, yaml, oracledb
from collections import Counter
from core.secrets import create_secret_provider

RUN_REPEAT = False  # True only for an approved disposable TARGET
cfg = yaml.safe_load(open("config/oracle_local_to_local.yaml"))
resolver = create_secret_provider(cfg)
tables = ["DEPARTMENTS", "DOCTORS", "PATIENTS", "APPOINTMENTS", "PAYMENTS"]
T = ",".join(repr(t) for t in tables)
U = "'HEALTH_READER','HEALTH_WRITER'"
R = "'HEALTH_READ_ROLE','HEALTH_WRITE_ROLE'"
failures = []

def connect(side, executor=False):
    c = cfg[side]["connection"]
    login = cfg["oracle_permission_migration"]["executor"] if executor else c
    return oracledb.connect(user=login["username"],
        password=resolver.resolve(login["password_secret"]),
        dsn=f'{c["host"]}:{c["port"]}/{c["service_name"]}')

def rows(conn, sql):
    with conn.cursor() as cur:
        cur.execute(sql)
        return [tuple(v.read() if isinstance(v, oracledb.LOB) else v for v in r)
                for r in cur.fetchall()]

def check(name, test):
    try:
        if not test():
            raise AssertionError("Mismatch")
        print("PASS -", name, flush=True)
    except Exception as exc:
        failures.append(name)
        print("FAIL -", name, exc, flush=True)

def same(s, t, sql, expected=None):
    a, b = Counter(rows(s, sql)), Counter(rows(t, sql))
    if a != b:
        print("SOURCE only:", a-b, "TARGET only:", b-a)
    return a == b and (expected is None or sum(b.values()) == expected)

def verify():
    with connect("source") as s, connect("target") as t, connect("target", True) as e:
        for table in tables:
            check(table + " exact data", lambda table=table: same(s,t,f"SELECT * FROM {table}",10))
        checks = {
            "Functional constraints": (f"""
                SELECT c.table_name,c.constraint_type,cc.column_name,cc.position,
                       c.search_condition_vc,r.table_name,rc.column_name,c.delete_rule,c.status,c.validated
                FROM user_constraints c
                LEFT JOIN user_cons_columns cc ON c.constraint_name=cc.constraint_name
                LEFT JOIN user_constraints r ON c.r_constraint_name=r.constraint_name AND c.r_owner=r.owner
                LEFT JOIN user_cons_columns rc ON r.constraint_name=rc.constraint_name AND cc.position=rc.position
                WHERE c.table_name IN ({T}) AND NOT (c.constraint_type='C'
                AND c.generated='GENERATED NAME' AND NVL(c.search_condition_vc,' ') LIKE '%IS NOT NULL')
            """, 15),
            "Column nullability": (f"SELECT table_name,column_name,nullable FROM user_tab_columns WHERE table_name IN ({T})", None),
            "Indexes and DESC": ("SELECT index_name,table_name,column_position,descend FROM user_ind_columns WHERE index_name IN ('IDX_DOCTOR_DEPT','IDX_PATIENT_NAME','IDX_APPT_DATE','IDX_PAYMENT_APPT')", 4),
            "Partitions": ("SELECT table_name,partition_name,partition_position,high_value FROM user_tab_partitions WHERE table_name='PAYMENTS'", 2),
            "Identity definition": ("SELECT table_name,column_name,generation_type FROM user_tab_identity_cols WHERE table_name='PATIENTS'", 1),
            "Table comments": ("SELECT table_name,comments FROM user_tab_comments WHERE comments IS NOT NULL AND table_name IN ('PATIENTS','PAYMENTS')", 2),
            "Column comments": ("SELECT table_name,column_name,comments FROM user_col_comments WHERE comments IS NOT NULL AND table_name IN ('PATIENTS','PAYMENTS')", 2),
            "Object grants": (f"SELECT grantee,table_name,privilege,grantable FROM user_tab_privs_made WHERE grantee IN ({U},{R})", 10),
            "Column grants": (f"SELECT grantee,table_name,column_name,privilege,grantable FROM user_col_privs_made WHERE grantee IN ({U},{R})", 1),
        }
        for name, (sql, count) in checks.items():
            check(name, lambda sql=sql,count=count: same(s,t,sql,count))
        obj = "SELECT object_name,object_type,status FROM user_objects WHERE generated='N'"
        check("All SOURCE objects preserved", lambda: set(rows(s,obj)).issubset(set(rows(t,obj))))
        check("Zero invalid TARGET objects", lambda: not rows(t,"SELECT object_name,object_type FROM user_objects WHERE status='INVALID'"))
        check("Zero compilation errors", lambda: not rows(t,"SELECT name,type,text FROM user_errors WHERE attribute='ERROR'"))
        for obj in ["V_PATIENT_APPOINTMENTS","V_PAYMENT_SUMMARY","MV_DOCTOR_APPOINTMENTS","SYN_PATIENTS","SYN_DOCTORS"]:
            check(obj, lambda obj=obj: same(s,t,f"SELECT * FROM {obj}"))
        check("Function", lambda: same(s,t,"SELECT FUNC_TOTAL_PAYMENT(1) FROM dual"))
        check("Package", lambda: same(s,t,"SELECT PKG_HEALTHCARE.GET_PATIENT_COUNT() FROM dual"))
        check("Two roles", lambda: len(rows(e,f"SELECT role FROM dba_roles WHERE role IN ({R})"))==2)
        check("Three memberships", lambda: same(s,e,f"SELECT grantee,granted_role FROM dba_role_privs WHERE grantee IN ({U})",3))
        check("Two CREATE SESSION grants", lambda: same(s,e,f"SELECT grantee,privilege FROM dba_sys_privs WHERE grantee IN ({U}) AND privilege='CREATE SESSION'",2))
        check("Two locked users", lambda: len(rows(e,f"SELECT username FROM dba_users WHERE username IN ({U}) AND account_status LIKE '%LOCKED%'"))==2)

def stop_on_failure():
    if failures:
        sys.exit("FAILED CHECKS: " + str(failures))

verify()
stop_on_failure()
if not RUN_REPEAT:
    print("READ-ONLY ACCEPTANCE CHECKS PASSED")
    sys.exit(0)

with connect("target") as t:
    with t.cursor() as cur:
        cur.execute("UPDATE DEPARTMENTS SET LOCATION='MDM_OVERWRITE_TEST' WHERE DEPARTMENT_ID=1")
        if cur.rowcount != 1:
            t.rollback()
            sys.exit("Overwrite test row not found")
    t.commit()

subprocess.run([sys.executable,"-m","migration_platform","--config",
                "config/oracle_local_to_local.yaml","--mode","full","--exit-after-run"],check=True)
verify()
stop_on_failure()

with connect("target") as t:
    check("Overwrite marker removed", lambda: rows(t,"SELECT COUNT(*) FROM DEPARTMENTS WHERE LOCATION='MDM_OVERWRITE_TEST'")[0][0]==0)
    def identity_test():
        max_id = rows(t,"SELECT MAX(PATIENT_ID) FROM PATIENTS")[0][0]
        try:
            with t.cursor() as cur:
                out = cur.var(oracledb.DB_TYPE_NUMBER)
                cur.execute("INSERT INTO PATIENTS (PATIENT_NAME) VALUES ('Identity acceptance test') RETURNING PATIENT_ID INTO :id",id=out)
                new_id = out.getvalue()[0]
                print("Generated patient ID:", new_id)
                return new_id > max_id
        finally:
            t.rollback()
    check("Identity generator", identity_test)
    check("Test row rolled back", lambda: rows(t,"SELECT COUNT(*) FROM PATIENTS")[0][0]==10)
stop_on_failure()
print("ALL ACCEPTANCE CHECKS PASSED")
'@ | python -
```

Expected: all 28 checks PASS; with repeat enabled, the overwrite marker disappears, generated patient ID exceeds the loaded maximum (recorded: 11), and PATIENTS remains at 10 rows. The constraint check excludes redundant generated NOT NULL rows and checks nullability separately (SOURCE 25 vs TARGET 29 raw constraint rows). This is a fixture-specific script adapted from the live checks; not a production or exhaustive object-definition test. It checks all non-generated SOURCE object names/types/status and all TARGET invalid objects; independent TARGET-only invalid objects may require separate assessment (Document 04).

## 9. Quick troubleshooting

| Issue | Action |
|---|---|
| SQL*Plus paste / ORA-00922 / ORA-03048 | One statement at a time; `/` on separate line for PL/SQL. |

| DESC index rejected | Verify current normalizer and catalog metadata; do not change DESC to ASC. |
